// 🔗 JOIN — attach the AI's score to each original job (matched by id)
const jobs = $('Pack Jobs for AI').first().json.jobs;
const matches = $input.first().json.output?.matches ?? [];
const list = arr => (Array.isArray(arr) ? arr.join(', ') : String(arr ?? ''));

return matches
  .filter(m => jobs[m.id])
  .map(m => {
    const job = jobs[m.id];
    return {
      json: {
        date_found: $now.toFormat('yyyy-MM-dd'),
        score: Number(m.score) || 0,
        title: job.title,
        company: job.company,
        location: job.location,
        lebanon_ok: m.lebanon_ok === true || m.lebanon_ok === 'true' ? 'YES' : 'NO',
        reason: m.reason,
        matched_skills: list(m.matched_skills),
        missing_skills: list(m.missing_skills),
        source: job.source,
        url: job.url,
      },
    };
  })
  .sort((a, b) => b.json.score - a.json.score);
